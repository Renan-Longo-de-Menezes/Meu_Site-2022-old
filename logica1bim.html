Algoritmo "MaiorMenor50"

Encontra o maior e o menor número entre 50 valores informados.

Algoritmo "MaiorMenor50"
Var
    numero, maior, menor, i: inteiro

Inicio
    Escreva ("Entre com 1° número")
    leia (numero)
    menor <- numero
    maior <- numero

    para i de 2 ate 50 faca
        Escreva ("Entre com o ", i, "° número:")
        leia(numero)

        se numero > maior entao
            maior <- numero
        fimse

        se numero < menor entao
            menor <- numero
        fimse
    fimpara

    Escreval("O maior número é:", maior)
    Escreval("O menor número é:", menor)

Algoritmo "VolumeDaEsfera"

Calcula o volume de uma esfera a partir do parâmetro π e do raio.

Algoritmo "VolumeDaEsfera"
Var
    volume, parametro, raio : real

Inicio
    // Entrada
    leia(parametro)
    leia(raio)

    // Processamento
    volume <- 4/3 * parametro * raio^3

    // Saída
    escreva (volume)
